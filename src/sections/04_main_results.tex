We begin with Lemma~\eqref{lem:central-newtons-formula-for-powers}, which gives Newton's interpolation formula for powers.
Applying Lemma~\eqref{lem:central-factorials-binomial-form} to this formula, for non-negative integers $n$ and $m$ and an arbitrary integer $t$, gives
\begin{align*}
  n^m
  &= \frac{\centralFactorial{(n-t)}{0}}{0!} \delta^{0} t^m
  + \sum_{k=1}^{m} \frac{n-t}{k} \binom{n+t+\frac{k}{2}-1}{k-1} \delta^{k} t^m \\
  &= t^m + \sum_{k=1}^{m} (n-t) \binom{n-t+\frac{k}{2}-1}{k-1} \frac{\delta^{k} t^m}{k}.
\end{align*}
Note that we isolate the term $\frac{\centralFactorial{(n-t)}{0}}{0!} \delta^{0} t^m$
to avoid division by zero.
By expanding the brackets, we have
\begin{align*}
  n^m
  = t^m + \sum_{k=1}^{m} \frac{\delta^{k} t^m}{k}
  \left[
    n\binom{n-t+\frac{k}{2}-1}{k-1}
    -
    t\binom{n-t+\frac{k}{2}-1}{k-1}
  \right].
\end{align*}
Notice that the factor $n\binom{n-t+\frac{k}{2}-1}{k-1}$ prevents us from direct
use of hockey-stick identity~\eqref{lem:generalized-hockey-stick-identity}.
The binomial decomposition in Lemma~\eqref{lem:binomial-decomposition}
handles this factor.
Substituting $r=-t+\frac{k}{2}-1$ and $m=k-1$ into
Lemma~\eqref{lem:binomial-decomposition} gives
\begin{align*}
  n \binom{n-t+\frac{k}{2} -1}{k-1}
  = k \binom{n-t+\frac{k}{2}-1}{k}
  + \left [t+\frac{k}{2} \right ]
  \binom{n-t+\frac{k}{2}-1}{k-1}.
\end{align*}
For clarity, applying Lemma~\eqref{lem:binomial-decomposition} directly gives
\begin{align*}
  n \tbinom{n-t+\frac{k}{2} -1}{k-1}
  &=
  k \tbinom{n-t+\frac{k}{2} -1}{k-1+1}
  - \left [-t+\tfrac{k}{2} -1-(k-1)\right ]
  \tbinom{n-t+\frac{k}{2} -1}{k-1} \\
  &= k \tbinom{n-t+\frac{k}{2}-1}{k}
  - \left [-t-\tfrac{k}{2} \right ]
  \tbinom{n-t+\tfrac{k}{2}-1}{k-1} \\
  &= k \tbinom{n-t+\tfrac{k}{2}-1}{k}
  + \left [t+\tfrac{k}{2} \right ]
  \tbinom{n-t+\tfrac{k}{2}-1}{k-1}.
\end{align*}
Thus,
\begin{align*}
  n^m
  = t^m
  + \sum_{k=1}^{m} \frac{\delta^{k} t^m}{k}
  \left[
    k \binom{n-t+\frac{k}{2}-1}{k}
    +
    \left[t+\frac{k}{2} \right]
    \binom{n-t+\frac{k}{2}-1}{k-1}
    -
    t \binom{n-t+\frac{k}{2}-1}{k-1}
  \right].
\end{align*}
Combining the terms involving $t\binom{n-t+\frac{k}{2}-1}{k-1}$ and simplifying the factors of $k$ gives
\begin{align*}
  n^m
  = t^m
  + \sum_{k=1}^{m}
  \left[
    \binom{n-t+\frac{k}{2}-1}{k}
    +
    \frac{1}{2}
    \binom{n-t+\frac{k}{2}-1}{k-1}
  \right] \delta^{k} t^m.
\end{align*}
We absorb the term $t^m$ into the sum to obtain
\begin{align*}
  n^m
  = \sum_{k=0}^{m}
  \left[
    \binom{n-t+\frac{k}{2}-1}{k}
    +
    \frac{1}{2}
    \binom{n-t+\frac{k}{2}-1}{k-1}
  \right] \delta^{k} t^m.
\end{align*}
We now simplify the binomial basis $\binom{n-t+\frac{k}{2}-1}{k}+\frac{1}{2}\binom{n-t+\frac{k}{2}-1}{k-1}$ to remove the fractional coefficient.
Set $a=n-t+\frac{k}{2}-1$.
\begin{align*}
  \binom{a}{k} + \frac{1}{2} \binom{a}{k-1}
  = \frac{1}{2}
  \left[
    2 \binom{a}{k} + \binom{a}{k-1}
  \right]
  = \frac{1}{2}
  \left[
    \binom{a}{k} + \binom{a+1}{k}
  \right].
\end{align*}
Therefore,
\begin{align*}
  n^m
  = \frac{1}{2}
  \sum_{k=0}^{m}
  \left[
    \binom{n-t+\frac{k}{2}-1}{k}
    +
    \binom{n-t+\frac{k}{2}}{k-1}
  \right] \delta^{k} t^m.
\end{align*}
The upper indices of the binomial coefficients may be fractional, because $\binom{n}{k}$ is a polynomial in $n$.
The identity involving the unsigned Stirling numbers of the first kind $\stirlingi{n}{k}$ makes this explicit:
\begin{align*}
  \binom{n}{k}
  = \sum_{j=0}^{k} \frac{(-1)^{k-j}}{k!} \stirlingi{k}{j} n^j.
\end{align*}

We can now apply the centered hockey-stick identity in Lemma~\eqref{lem:centered-hockey-stick-identity} to obtain a closed formula for sums of powers:
\begin{align*}
  \KnuthRFoldSum{1}{n}{m}
  = \frac{1}{2}
  \sum_{k=0}^{m}
  \left[
    \sum_{j=1}^{n} \binom{j-t+\frac{k}{2}-1}{k}
    +
    \sum_{j=1}^{n} \binom{j-t+\frac{k}{2}}{k-1}
  \right] \delta^{k} t^m.
\end{align*}
Applying Lemma~\eqref{lem:centered-hockey-stick-identity} gives
\begin{align*}
  \begin{cases}
    \sum_{j=1}^{n} \binom{j-t+\frac{k}{2}-1}{k} &= \binom{n -t + \frac{k}{2} +0}{k+1} - \binom{0 -t + \tfrac{k}{2}}{k+1}, \\
    \sum_{j=1}^{n} \binom{j-t+\frac{k}{2}}{k-1} &= \binom{n -t + \frac{k}{2} +1}{k+1} - \binom{1 -t + \tfrac{k}{2}}{k+1}.
  \end{cases}
\end{align*}
We obtain the following closed formula for ordinary sums of powers.
\begin{proposition}[Ordinary sums of powers]
  \label{prop:ordinary-sums-of-powers}
  Let $n$ and $m$ be nonnegative integers, and let $t$ be an arbitrary integer.
  \begin{align*}
    \KnuthRFoldSum{1}{n}{m}
    = \frac{1}{2}
    \sum_{k=0}^{m}
    \left[
      \binom{n -t + \frac{k}{2} +0}{k+1} + \binom{n -t + \frac{k}{2} +1}{k+1} - \binom{0 -t + \tfrac{k}{2}}{k+1}
      - \binom{1 -t + \tfrac{k}{2}}{k+1}
    \right] \delta^{k} t^m.
  \end{align*}
\end{proposition}

Similarly, setting $r=0$ and $r=1$ in Lemma~\eqref{lem:centered-hockey-stick-identity} gives
\begin{align*}
  \begin{cases}
    \sum_{j=1}^{n} \binom{j -t + \frac{k}{2} +0}{k+1} &= \binom{n -t + \frac{k}{2} +1}{k+2} - \binom{1 -t + \tfrac{k}{2}}{k+2}, \\
    \sum_{j=1}^{n} \binom{j -t + \frac{k}{2} +1}{k+1} &= \binom{n -t + \frac{k}{2} + 2}{k+2} - \binom{2 -t + \tfrac{k}{2}}{k+2}.
  \end{cases}
\end{align*}
We can now state the formula for double sums of powers.
\begin{proposition}[Double sums of powers]
  \label{prop:double-sums-of-powers}
  Let $n$ and $m$ be nonnegative integers, and let $t$ be an arbitrary integer.
  \begin{align*}
    \KnuthRFoldSum{2}{n}{m}
    &= \frac{1}{2}
    \sum_{k=0}^{m}
    \Bigg[
      \binom{n -t + \frac{k}{2} +1}{k+2}
      +\binom{n -t + \frac{k}{2} + 2}{k+2} \\
      &-
      \Bigg(
        \binom{0 -t + \tfrac{k}{2}}{k+1}
        +\binom{1 -t + \tfrac{k}{2}}{k+1}
      \Bigg) \KnuthRFoldSum{1}{n}{0} \\
      &-
      \Bigg(
        \binom{1 -t + \tfrac{k}{2}}{k+2}
        +\binom{2 -t + \tfrac{k}{2}}{k+2}
      \Bigg) \KnuthRFoldSum{0}{n}{0}
    \Bigg] \delta^{k} t^m.
  \end{align*}
\end{proposition}

The general case is as follows.
\begin{theorem}[Multifold sums of powers]
  \label{thm:multifold-sums-of-powers}
  Let $r$, $n$, and $m$ be nonnegative integers, and let $t$ be an arbitrary integer.
  \begin{align*}
    \KnuthRFoldSum{r+1}{n}{m}
    &= \frac{1}{2}
    \sum_{k=0}^{m}
    \Bigg[
      \binom{n-t+\frac{k}{2}+r}{k+r+1}
      +
      \binom{n-t+\frac{k}{2}+r+1}{k+r+1} \\
      &-
      \sum_{s=0}^{r}
      \Bigg(
        \binom{s-t+\frac{k}{2}}{k+s+1}
        +
        \binom{s-t+\frac{k}{2}+1}{k+s+1}
      \Bigg) \KnuthRFoldSum{r-s}{n}{0}
    \Bigg] \delta^{k} t^{m}.
  \end{align*}
\end{theorem}
