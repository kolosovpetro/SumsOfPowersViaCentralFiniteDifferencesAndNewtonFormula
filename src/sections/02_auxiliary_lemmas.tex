This section introduces the lemmas, definitions, and conventions used in the main results.
We use Donald Knuth's notation for \emph{multifold sums of powers} as given in~\cite{knuth1993johann} because it is concise.
\begin{definition} [Multifold sums of powers]
  \label{def:multifold-sums-of-powers-recurrence}
  For non-negative integers $r$, $n$, and $m$, define the \emph{multifold sums of powers} recursively by
  \begin{align*}
    \KnuthRFoldSum{0}{n}{m}   &= n^m, \\
    \KnuthRFoldSum{1}{n}{m}   &= \KnuthRFoldSum{0}{1}{m}
    + \KnuthRFoldSum{0}{2}{m} + \cdots + \KnuthRFoldSum{0}{n}{m}, \\
    \KnuthRFoldSum{r+1}{n}{m} &= \KnuthRFoldSum{r}{1}{m}
    + \KnuthRFoldSum{r}{2}{m} + \cdots + \KnuthRFoldSum{r}{n}{m}.
  \end{align*}
\end{definition}

We define the falling factorial notation as follows.
\begin{definition} [Falling factorials] For integers $n$ and $k$, define the \emph{falling factorial} by
  \begin{align*}
    \fallingFactorial{n}{k} =
    \begin{cases}
      1, & \mathrm{if \; } k = 0, \\
      n(n-1)(n-2)\cdots(n-k+1) = \prod_{j=0}^{k-1} (n-j), & \mathrm{if \; } k > 0, \\
      0, & \mathrm{otherwise}.
    \end{cases}
  \end{align*}
\end{definition}
We define the central factorial notation as follows.
\begin{definition}[Central factorials]
  \label{def:central-factorials}
  For integers $n$ and $k$, define the \emph{central factorial} by
  \begin{align*}
    \centralFactorial{n}{k} =
    \begin{cases}
      0, \quad & \mathrm{if \; } k < 0, \\
      1, \quad & \mathrm{if \; } k = 0, \\
      n \left( n + \frac{k}{2} -1 \right)\left( n + \frac{k}{2} -2 \right)
      \cdots \left( n - \frac{k}{2} +1 \right)
      = n \fallingFactorial{n+\frac{k}{2}-1}{k-1}, \quad & \mathrm{if \; } k>0,
    \end{cases}
  \end{align*}
  where
  $\fallingFactorial{n+\frac{k}{2}-1}{k-1}
  = \prod_{j=0}^{k-2} \left( n+\frac{k}{2}-1 -j \right)$
  is a falling factorial.
\end{definition}
For further discussion of generalized factorials, see J. F. Steffensen's treatment in~\cite{steffensen1933definition}.

\begin{lemma}[Central Newton's formula]
  Let $f$ be a function defined on the integers, and let $x$ and $t$ be arbitrary integers.
  \label{lem:central-newtons-formula}
  \begin{align*}
    f(x)
    = \sum_{k=0}^{\infty} \frac{\centralFactorial{(x-t)}{k}}{k!} \delta^{k} f(t).
  \end{align*}
  The $k$th central finite difference is defined by
  \begin{align*}
    \delta^{k} f(t) = \tsum_{j=0}^{k} (-1)^j \tbinom{k}{j} f\!\left(t+\tfrac{k}{2} - j\right),
  \end{align*}
  and $\centralFactorial{(x-t)}{k}$ denotes the central factorial defined in Definition~\eqref{def:central-factorials}.
  \begin{proof}
    Proofs of the central Newton formula are given in~\cite[p.~462]{butzer1989central}, in~\cite[Section 2, Equation (19)]{steffensen1927interpolation}, and in~\cite[Section 6, Equation (21)]{riordan1968combinatorial}.
  \end{proof}
\end{lemma}
The specialization to powers gives the following lemma.
\begin{lemma}[Newton's formula for powers]
  \label{lem:central-newtons-formula-for-powers}
  Let $n$ and $m$ be nonnegative integers, and let $t$ be an arbitrary integer.
  \begin{align*}
    n^m = \sum_{k=0}^{m} \frac{\centralFactorial{(n-t)}{k}}{k!} \delta^{k} t^m.
  \end{align*}
  \begin{proof}
    This is the special case of Lemma~\eqref{lem:central-newtons-formula} with $f(n) = n^m$.
  \end{proof}
\end{lemma}
The central factorial identity has the following binomial form.
\begin{lemma}[Central factorials binomial form]
  \label{lem:central-factorials-binomial-form}
  For integers $n$ and nonzero $k$,
  \begin{align*}
    \frac{\centralFactorial{n}{k}}{k!}
    = \frac{n}{k} \binom{n+\frac{k}{2}-1}{k-1}.
  \end{align*}
  \begin{proof}
    We have
    \begin{align*}
      \frac{\centralFactorial{n}{k}}{k!}
      &= \frac{n}{k!}\fallingFactorial{n+\frac{k}{2}-1}{k-1}
      = \frac{n}{k (k-1)!} \fallingFactorial{n+\frac{k}{2}-1}{k-1}
      = \frac{n}{k} \tbinom{n+\frac{k}{2}-1}{k-1},
    \end{align*}
    by the falling-factorial identity
    $\frac{\fallingFactorial{x}{n}}{n!} = \tbinom{x}{n}$.
  \end{proof}
\end{lemma}
The following identity separates a factor of $n$ from a binomial coefficient.
\begin{lemma}[Binomial decomposition]
  \label{lem:binomial-decomposition}
  Let $r$, $n$, and $m$ be nonnegative integers.
  \begin{align*}
    n \binom{n+r}{m} = (m+1) \binom{n+r}{m+1} - (r-m) \binom{n+r}{m}.
  \end{align*}
  \begin{proof}
    Expanding the right-hand side gives
    \begin{align*}
      n \tbinom{n+r}{m}
      = m \tbinom{n+r}{m+1} + \tbinom{n+r}{m+1} - r \tbinom{n+r}{m}
      + m \tbinom{n+r}{m}.
    \end{align*}
    Recall the extraction property of binomial coefficients
    $\tbinom{n}{k+1} = \frac{n-k}{k+1} \tbinom{n}{k}$.
    The extraction property then gives
    \begin{align*}
      \tbinom{n+r}{m+1} = \frac{n+r-m}{m+1} \tbinom{n+r}{m}.
    \end{align*}
    Thus,
    \begin{align*}
      n \tbinom{n+r}{m}
      &= m \frac{n+r-m}{m+1} \tbinom{n+r}{m}
      + \frac{n+r-m}{m+1} \tbinom{n+r}{m}
      - r \tbinom{n+r}{m} + m \tbinom{n+r}{m}.
    \end{align*}
    Factoring out the binomial coefficient $\tbinom{n+r}{m}$ gives
    \begin{align*}
      n \tbinom{n+r}{m}
      &= \tbinom{n+r}{m} \left[ m \frac{n+r-m}{m+1} + \frac{n+r-m}{m+1} - r + m \right] \\
      &= \tbinom{n+r}{m} \left[ (m+1) \frac{n+r-m}{m+1} - r + m \right] \\
      &= \tbinom{n+r}{m}n.
    \end{align*}
    This completes the proof.
  \end{proof}
\end{lemma}
We also use the generalized hockey-stick identity.
\begin{lemma}[Generalized hockey-stick identity]
  \label{lem:generalized-hockey-stick-identity}
  For arbitrary integers $a$, $b$, and $j$,
  \begin{align*}
    \sum_{k=a}^{b} \binom{k}{j} = \binom{b+1}{j+1} - \binom{a}{j+1}.
  \end{align*}
  \begin{proof}
    We have
    $\tsum_{k=a}^{b} \tbinom{k}{j}
    = \tbinom{a}{j} + \tbinom{a+1}{j} + \cdots + \tbinom{b}{j}
    = \left( \tsum_{k=0}^{b} \tbinom{k}{j} \right) - \left( \tsum_{k=0}^{a-1} \tbinom{k}{j} \right)$.
    The hockey-stick identity $\tsum_{k=0}^{n} \tbinom{k}{j} = \tbinom{n+1}{j+1}$ therefore gives
    \begin{align*}
      \tsum_{k=a}^{b} \tbinom{k}{j}
      = \left( \tsum_{k=0}^{b} \tbinom{k}{j} \right) - \left( \tsum_{k=0}^{a-1} \tbinom{k}{j} \right)
      = \tbinom{b+1}{j+1} - \tbinom{a}{j+1}.
    \end{align*}
    This completes the proof.
  \end{proof}
\end{lemma}
The centered version is as follows.
\begin{lemma}[Centered hockey-stick identity]
  \label{lem:centered-hockey-stick-identity}
  Let $n$ be a positive integer, and let $t$, $k$, and $r$ be arbitrary integers.
  \begin{align*}
    \sum_{j=1}^{n} \binom{j -t + \frac{k}{2} +r}{k}
    = \binom{n -t + \frac{k}{2} + r +1}{k+1} - \binom{1 -t + \tfrac{k}{2} +r}{k+1}.
  \end{align*}
  \begin{proof}
    Substituting $a=1 -t + \tfrac{k}{2} +r$ and $b=n-t+ \tfrac{k}{2} +r$ into Lemma~\eqref{lem:generalized-hockey-stick-identity} gives
    \begin{align*}
      \tsum_{j=1}^{n} \tbinom{j -t + \tfrac{k}{2} +r}{k} =
      \tsum_{j=1 -t + \tfrac{k}{2} +r}^{n -t + \tfrac{k}{2} +r} \tbinom{j}{k}
      = \tbinom{n -t + \tfrac{k}{2} + r +1}{k+1} - \tbinom{1 -t + \tfrac{k}{2} +r}{k+1}.
    \end{align*}
    This completes the proof.
  \end{proof}
\end{lemma}
The following formula gives the multifold sum of zeroth powers.
\begin{lemma}[Multifold sums of zero powers]
  \label{lem:multifold-sum-of-zero-powers}
  Let $r$ and $n$ be nonnegative integers.
  \begin{align*}
    \KnuthRFoldSum{r}{n}{0} = \binom{r+n-1}{r}.
  \end{align*}
  \begin{proof}
    \leavevmode\par
    \begin{enumerate}
      \item When $r=0$, Definition~\eqref{def:multifold-sums-of-powers-recurrence} gives $\KnuthRFoldSum{0}{n}{0} = 1$.
      \item When $r=1$, we have
        $\KnuthRFoldSum{1}{n}{0}
        = \tsum_{k=1}^{n} 1 = \tbinom{n}{1}$.
      \item When $r=2$, we have
        $\KnuthRFoldSum{2}{n}{0}
        = \tsum_{k=1}^{n} \tbinom{k}{1}
        = \tbinom{n+1}{2}$.
      \item When $r=3$, we have
        $\KnuthRFoldSum{3}{n}{0}
        = \tsum_{k=1}^{n} \tbinom{k+1}{2}
        = \tbinom{n+2}{3}$.
      \item For general $r$, the hockey-stick identity gives
        $\KnuthRFoldSum{r}{n}{0}
        =  \tsum_{k=1}^{n} \KnuthRFoldSum{r-1}{k}{0}
        =  \tsum_{k=1}^{n} \tbinom{k+r-2}{r-1}
        = \tbinom{r+n-1}{r}$.
    \end{enumerate}
    This completes the proof.
  \end{proof}
\end{lemma}
Throughout, we use the following convention.
\begin{convention} For all $x$,
  \begin{align*}
    x^0 = 1.
  \end{align*}
\end{convention}
Donald Knuth discusses the convention $x^0=1$ for all $x$, including zero, in Concrete Mathematics, as described in~\cite[p.~162]{graham1994concrete} and in his notes in~\cite{knuth1992notesnotation}.
