(* ================================================================ *)
(*  Validates Proposition: Multifold Binomial sums of powers        *)
(* ================================================================ *)

ClearAll[
  KnuthRFoldSum,
  CentralDifference,
  ValidateMultifoldBinomialSumsOfPowers
];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                  *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];

(* --------------------------------------------------------------- *)
(*  Multifold sums of powers                                       *)
(* --------------------------------------------------------------- *)

KnuthRFoldSum[0, n_, m_] := n^m;

KnuthRFoldSum[r_, n_, m_] :=
  Sum[KnuthRFoldSum[r - 1, k, m], {k, 1, n}];

(* --------------------------------------------------------------- *)
(*  Central finite difference                                      *)
(* --------------------------------------------------------------- *)

CentralDifference[t_, m_, k_] :=
  Sum[
    (-1)^j Binomial[k, j] (t + k/2 - j)^m,
    {j, 0, k}
  ];

(* --------------------------------------------------------------- *)
(*  Main identity                                                  *)
(* --------------------------------------------------------------- *)

MultifoldBinomialSumsOfPowers[r_, n_, m_, t_] :=
  1/2 Sum[
    (
      Binomial[n - t + k/2 + r, k + r + 1]
      + Binomial[n - t + k/2 + r + 1, k + r + 1]
      - Sum[
          (Binomial[s - t + k/2, k + s + 1] + Binomial[s - t + k/2 + 1, k + s + 1])
          * Binomial[r - s + n - 1, r - s],
          {s, 0, r}
        ]
    )
    * CentralDifference[t, m, k],
    {k, 0, m}
  ];

(* --------------------------------------------------------------- *)
(*  Validation routine                                             *)
(* --------------------------------------------------------------- *)

ValidateMultifoldBinomialSumsOfPowers[max_] :=
  Flatten[
    Table[
      FullSimplify[
        KnuthRFoldSum[r + 1, n, m]
        -
        MultifoldBinomialSumsOfPowers[r, n, m, t]
      ],
      {r, 0, max},
      {n, 1, max},
      {m, 0, max},
      {t, -max, max}
    ]
  ];

(* --------------------------------------------------------------- *)
(*  Run test                                                       *)
(* --------------------------------------------------------------- *)

ValidateMultifoldBinomialSumsOfPowers[5]
